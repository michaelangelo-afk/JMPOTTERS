## CODE REFERENCE — THE BLUE MODAL (the intended checkout UX)

This is **the modal you're actually supposed to see** when you click

### Entry point — `proceedToCheckout()` (`app.js` line ~1870)

```js
function proceedToCheckout() {
    if (isProcessingCheckout) return;

    const cart = getCart();
    if (cart.length === 0) {
        showNotification('Your cart is empty', 'warning');
        return;
    }

    const user = safeParseJSON('jmpotters_user', null);

    if (user && user.address && user.city && user.state) {
        // v15: open the in-app FULL-SCREEN payment modal (replaces the
        // older checkout.html redirect). The modal contains UBA + OPay
        // account cards, a dropdown for selecting which account to copy,
        // receipt-upload dropzone, and place-order button.
        try {
            if (typeof showCheckoutModal === 'function') {
                showCheckoutModal();   // <-- THE BLUE MODAL opens here
                return;
            }
            if (typeof window.showCheckoutModal === 'function') {
                window.showCheckoutModal();
                return;
            }
        } catch(_e) { /* silent */ }
        // Last-resort fallback: redirect to checkout.html.
        window.location.href = 'checkout.html';
        return;
    }

    if (user && (!user.address || !user.city || !user.state)) {
        showNotification('Please complete your profile before checkout', 'warning');
        sessionStorage.setItem('checkoutRedirect', window.location.href);
        window.location.href = 'register.html';
        return;
    }

    sessionStorage.setItem('checkoutRedirect', window.location.href);
    showNotification('Please create an account to complete your purchase', 'info');
    window.location.href = 'register.html';
}
```

### Bank-account copy helper (post-IIFE, `app.js` line ~2540+)

```js
window.JMPOTTERS_copyBankAccount = function (num, btnEl) {
    if (!num || !btnEl) return;
    var original = btnEl.dataset.v16Original || btnEl.innerHTML;
    btnEl.dataset.v16Original = original;

    var setState = function (kind) {
        if (kind === 'ok') {
            btnEl.innerHTML = '✓ Copied';
            btnEl.style.background = 'linear-gradient(135deg,#10b981,#059669)';
            btnEl.style.boxShadow = '0 0 14px rgba(16,185,129,0.5)';
            btnEl.disabled = true;
        } else if (kind === 'err') {
            btnEl.innerHTML = '✗ Try again';
            btnEl.style.background = 'linear-gradient(135deg,#ef4444,#b91c1c)';
            btnEl.disabled = true;
        } else {
            btnEl.innerHTML = original;
            btnEl.disabled = false;
        }
    };
    var fallbackCopy = function (text) { /* textarea + execCommand */ };

    if (navigator.clipboard && navigator.clipboard.writeText) {
        navigator.clipboard.writeText(num).then(
            function () { setState('ok'); setTimeout(function(){setState('reset');}, 1800); },
            function () { if (fallbackCopy(num)) setState('ok'); else setState('err'); }
        );
    } else if (fallbackCopy(num)) {
        setState('ok');
    } else {
        setState('err');
    }
};

window.JMPOTTERS_toggleBankCard = function (selectEl) {
    var which = selectEl && selectEl.value;
    var uba  = document.getElementById('v16UbaCard');
    var opay = document.getElementById('v16OpayCard');
    if (uba)  uba.style.display  = (which === 'uba')  ? 'block' : 'none';
    if (opay) opay.style.display = (which === 'opay') ? 'block' : 'none';
};
```
